## 2. Условие задачи

Вариант 22. Структура «автобус» (поля: модель (название), длина, высота, максимальное число пассажиров). Создается массив (вектор) автобусов размерности n. Массив сохраняется в файле и читается из файла в текстовом или двоичном режимах. Необходимо сравнить размеры файлов с данными в текстовом и двоичном режимах.

Режимы приложения задаются параметрами командной строки: сначала чтение или запись (`read` / `write`), затем текстовый или двоичный режим (`text` / `binary`), далее имя файла, например: `write binary bus.dat` — запись в двоичном режиме в файл bus.dat; `read text bus.txt` — чтение в текстовом режиме из файла bus.txt. В режиме записи данные вводятся с клавиатуры, в режиме чтения — читаются из файла и выводятся на консоль. В работе используются шаблон `vector` и класс `string`; чтение и запись структурной переменной выполняются отдельными функциями для текстового и двоичного режимов.

## 3. Текст программы с комментариями

```cpp
// Лабораторная работа № 2. Ввод-вывод в файлы
// Вариант 22. Структура «автобус» (модель, длина, высота, максимальное число пассажиров)
//
// Режимы приложения задаются параметрами командной строки:
//   lab2_bus write text   bus.txt   - запись в файл в текстовом режиме
//   lab2_bus write binary bus.dat   - запись в файл в двоичном режиме
//   lab2_bus read  text   bus.txt   - чтение из файла в текстовом режиме
//   lab2_bus read  binary bus.dat   - чтение из файла в двоичном режиме

#include <iostream>
#include <fstream>   // ifstream, ofstream
#include <string>
#include <vector>
#include <limits>    // numeric_limits (для очистки буфера ввода)
using namespace std;

// Структура «автобус»
struct BUS
{
	string Model;     // модель (название)
	double Length;    // длина, м
	double Height;    // высота, м
	int MaxPassengers; // максимальное число пассажиров
};

// ---------- Функции для структурной переменной (текстовый режим) ----------

// Запись одного автобуса в текстовый поток:
// модель - на отдельной строке, числа - в одной строке через пробел
void write_bus_text(ostream& out, const BUS& b)
{
	out << b.Model << endl;
	out << b.Length << ' ' << b.Height << ' ' << b.MaxPassengers << endl;
}

// Чтение одного автобуса из текстового потока
bool read_bus_text(istream& in, BUS& b)
{
	getline(in, b.Model);                       // строка с моделью (может содержать пробелы)
	in >> b.Length >> b.Height >> b.MaxPassengers; // числа читаем операцией извлечения
	in.ignore(numeric_limits<streamsize>::max(), '\n'); // пропускаем остаток строки (символ конца строки)
	return static_cast<bool>(in);               // true, если чтение прошло без ошибок
}

// ---------- Функции для структурной переменной (двоичный режим) ----------

// Запись одного автобуса в двоичный поток:
// строка записывается вместе с завершающим нулём (признак конца строки),
// числа записываются в «сыром» виде - как они хранятся в памяти
void write_bus_bin(ostream& out, const BUS& b)
{
	out.write(b.Model.c_str(), b.Model.size() + 1); // +1 - вместе с символом '\0'
	out.write(reinterpret_cast<const char*>(&b.Length), sizeof(b.Length));
	out.write(reinterpret_cast<const char*>(&b.Height), sizeof(b.Height));
	out.write(reinterpret_cast<const char*>(&b.MaxPassengers), sizeof(b.MaxPassengers));
}

// Чтение одного автобуса из двоичного потока
bool read_bus_bin(istream& in, BUS& b)
{
	getline(in, b.Model, '\0');                 // читаем строку до нулевого символа (он извлекается из потока)
	in.read(reinterpret_cast<char*>(&b.Length), sizeof(b.Length));
	in.read(reinterpret_cast<char*>(&b.Height), sizeof(b.Height));
	in.read(reinterpret_cast<char*>(&b.MaxPassengers), sizeof(b.MaxPassengers));
	return static_cast<bool>(in);
}

// Печать массива автобусов в выходной поток (консоль)
void print_buses(ostream& out, const vector<BUS>& Buses)
{
	for (const auto& b : Buses)
		out << b.Model << ". Длина: " << b.Length << " м, высота: " << b.Height
		    << " м, вместимость: " << b.MaxPassengers << " пасс." << endl;
}

int main(int argc, const char* argv[])
{
	// Должно быть ровно 4 параметра: имя программы, режим, способ открытия, имя файла
	if (argc != 4) {
		cerr << "Error: Use four parameters (write|read text|binary filename)\n";
		return 5;
	}
	const string mode(argv[1]);      // режим работы: write или read
	const string open_mode(argv[2]); // режим открытия файла: text или binary
	const string file_name(argv[3]); // имя файла

	if (open_mode != "text" && open_mode != "binary") {
		cerr << "Error: second parameter must be text or binary\n";
		return 5;
	}

	vector<BUS> Buses; // массив (вектор) автобусов
	int n;             // число элементов

	if (mode == "write")
	{
		// ----- Ввод данных с клавиатуры -----
		cout << "n="; cin >> n;
		cin.ignore(numeric_limits<streamsize>::max(), '\n'); // убираем из потока символ конца строки после n
		Buses.resize(n);
		for (int i = 0; i < n; i++)
		{
			cout << "Bus N=" << (i + 1) << endl;
			cout << "Model: ";        getline(cin, Buses[i].Model);
			cout << "Length (m): ";   cin >> Buses[i].Length;
			cout << "Height (m): ";   cin >> Buses[i].Height;
			cout << "Max passengers: "; cin >> Buses[i].MaxPassengers;
			cin.ignore(numeric_limits<streamsize>::max(), '\n'); // готовим поток к следующему getline
		}

		// ----- Запись в файл -----
		if (open_mode == "text")
		{
			ofstream fout(file_name);        // текстовый режим (по умолчанию)
			if (!fout) { cerr << "Error, can't open file" << endl; return 1; }
			fout << n << endl;               // число элементов
			for (const auto& b : Buses) write_bus_text(fout, b);
			fout.close();
		}
		else
		{
			ofstream fout(file_name, ios::binary); // двоичный режим
			if (!fout) { cerr << "Error, can't open file" << endl; return 1; }
			fout.write(reinterpret_cast<const char*>(&n), sizeof(n));
			for (const auto& b : Buses) write_bus_bin(fout, b);
			fout.close();
		}
	}
	else if (mode == "read")
	{
		// ----- Чтение из файла -----
		if (open_mode == "text")
		{
			ifstream fin(file_name);
			if (!fin) { cout << endl << "Error, file not found" << endl; return 1; }
			fin >> n;
			fin.ignore(numeric_limits<streamsize>::max(), '\n'); // пропускаем конец строки после n
			Buses.resize(n);
			for (auto& b : Buses) read_bus_text(fin, b);
			fin.close();
		}
		else
		{
			ifstream fin(file_name, ios::binary);
			if (!fin) { cout << endl << "Error, file not found" << endl; return 1; }
			fin.read(reinterpret_cast<char*>(&n), sizeof(n));
			Buses.resize(n);
			for (auto& b : Buses) read_bus_bin(fin, b);
			fin.close();
		}
		// Печать данных, прочитанных из файла
		print_buses(cout, Buses);
	}
	else
	{
		cerr << "Error: first parameter must be write or read\n";
		return 5;
	}
	return 0;
}
```

## 5. Выводы по работе

В ходе работы разработана программа на языке Си++, которая сохраняет вектор структур «автобус» в файл и читает его из файла в текстовом и двоичном режимах; режим работы задается параметрами командной строки. Для обмена с файлами использованы классы `ofstream` и `ifstream`, а для хранения данных — шаблон `vector` и класс `string`.

**Сравнение файлов.** В текстовом режиме числа записываются в виде символов, поэтому размер зависит от количества цифр в значениях: для трех автобусов файл bus.txt занял 91 байт. В двоичном режиме каждое число записывается в том виде, в котором оно хранится в памяти (`int` — 4 байта, `double` — 8 байт), а строка — вместе с завершающим нулевым символом, поэтому размер записи не зависит от значений: файл bus.dat занял 120 байт. Для рассмотренных данных (короткие числа вида 11.5, 2.9, 43) текстовый файл получился компактнее, так как под число `double` в двоичном файле всегда отводится 8 байт. При значениях с большим числом цифр (например, 11.4999999) преимущество переходит к двоичному файлу.

Достоинства текстового режима: файл можно открыть и отредактировать в любом текстовом редакторе, формат переносим между платформами. Достоинства двоичного режима: фиксированный размер числовых полей, отсутствие преобразования числа в строку и обратно, поэтому чтение и запись выполняются быстрее и без потери точности. Следует учитывать, что в текстовом режиме в ОС Windows символ конца строки записывается как пара `\r\n`, что дополнительно увеличивает размер файла.

## Ответы на вопросы для самоконтроля

**1. Режимы открытия файлов.** `ios::in` — чтение; `ios::out` — запись; `ios::ate` — при открытии перейти в конец файла; `ios::app` — дописывать данные в конец файла; `ios::trunc` — очистить существующий файл; `ios::binary` — двоичный режим; `ios::nocreate` — не создавать новый файл; `ios::noreplace` — не открывать существующий файл (последние два есть не во всех реализациях).

**2. Отличия текстового и двоичного режимов.** В текстовом режиме данные записываются как последовательность символов, при этом в Windows выполняется преобразование `\n` ↔ `\r\n`; в двоичном режиме данные записываются «как есть», в том виде, в котором хранятся в памяти, без преобразований.

**3. Режимы, в которых файл должен существовать.** `ios::in` (а также `ios::nocreate`).

**4. Режимы, в которых файл создается заново.** `ios::out` (а также `ios::trunc`) — существующий файл очищается; `ios::app` и `ios::ate` содержимое не очищают.
