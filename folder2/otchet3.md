2.1. Условие задачи
Дан класс Vector, задающий вектор размерности n. Поля класса: указатель на динамический массив элементов типа int и число элементов (тип int). Класс включает конструктор без параметров (пустой вектор), конструктор на основе обычного массива, конструктор копирования, конструктор перемещения и деструктор.
Перегрузить операции [], = (с копированием), = (с перемещением), << (вставка в поток) и >> (извлечение из потока). Показать разницу между конструкторами копирования и перемещения и между присваиванием с копированием и перемещением. Код класса разместить в заголовочном файле и файле реализации. Входные данные читать из input.txt, результаты выводить в output.txt.
Индивидуальное задание (вариант 22): перегрузить операцию < для двух векторов (первый операнд Vector, второй операнд Vector, результат bool). Сравнение проводится последовательно по элементам. Результат истинно, если элемент первого операнда меньше элемента второго. Если элемент первого операнда больше, результат ложь. При равенстве происходит переход к следующей паре элементов. Если в каком-либо операнде элементы закончились, результат ложь.
2.2. Обоснование способа перегрузки
• operator< определён внутри класса (как метод). Первый операнд — объект класса Vector, поэтому внешняя функция не нужна. Метод имеет прямой доступ к полям p и n, а один параметр (второй вектор) соответствует бинарной операции.
• operator[] и operator= обязательно являются методами класса (так требует язык).
• operator<< и operator>> определены вне класса как дружественные функции. Первый операнд у них — поток (ostream / istream), а не объект Vector, поэтому методом класса они быть не могут. Дружественность нужна, чтобы функции имели доступ к закрытым полям.

// Vector.h - заголовочный файл класса Vector (вариант 22, часть 1)
#pragma once
#include <iostream>

class Vector // Класс-вектор (элементы типа int)
{
    int *p = nullptr; // Указатель на динамический массив (элементы вектора)
    int n = 0;        // Размерность вектора (число элементов)
public:
    Vector();                         // Конструктор без параметров: пустой вектор (n = 0)
    Vector(const int *arr, int n);    // Конструктор на основе обычного массива размерности n
    Vector(const Vector &V);          // Конструктор копирования
    Vector(Vector &&V) noexcept;      // Конструктор перемещения
    ~Vector();                        // Деструктор

    int size() const { return n; }            // Размерность вектора
    const int *data() const { return p; }     // Адрес буфера (нужен для демонстрации копирования/перемещения)

    int &operator[](int index);               // Обращение к элементу по индексу (чтение и запись)
    const int &operator[](int index) const;   // Обращение к элементу константного вектора

    Vector &operator=(const Vector &V);       // Присваивание с копированием
    Vector &operator=(Vector &&V) noexcept;   // Присваивание с перемещением

    // Задание 22: операция < (Vector < Vector), результат bool.
    // Функция класса: первый операнд - объект класса, поэтому ее можно сделать членом класса
    bool operator<(const Vector &V) const;

    // Вставка в поток (cout или файл ostream) и извлечение из потока (cin или файл istream).
    // Первый операнд - поток, а не объект класса, поэтому функции дружественные (вне класса)
    friend std::ostream &operator<<(std::ostream &os, const Vector &V);
    friend std::istream &operator>>(std::istream &is, Vector &V);
};

// Vector.cpp - файл реализации класса Vector (вариант 22, часть 1)
#include "Vector.h"
#include <iostream>
using namespace std;

// Конструктор без параметров: задает "пустой" вектор
Vector::Vector()
{
    p = nullptr;
    n = 0;
    cout << "  [Vector()]" << endl;
}

// Конструктор на основе обычного массива
Vector::Vector(const int *arr, int n)
{
    this->n = n;                  // Задаем число элементов
    this->p = new int[n];         // Выделяем память
    for (int i = 0; i < n; i++)   // Копируем элементы из обычного массива
        this->p[i] = arr[i];
    cout << "  [Vector(int*, n) n=" << n << "]" << endl;
}

// Конструктор копирования: создает НЕЗАВИСИМУЮ копию (новая память + копирование элементов)
Vector::Vector(const Vector &V)
{
    n = V.n;
    p = new int[n];               // Выделяем свою собственную память
    for (int i = 0; i < n; i++)
        p[i] = V.p[i];            // Копируем каждый элемент
    cout << "  [Vector(const Vector&) - КОПИРОВАНИЕ n=" << n << "]" << endl;
}

// Конструктор перемещения: "забирает" память у временного/перемещаемого объекта, ничего не копируя
Vector::Vector(Vector &&V) noexcept : p(V.p), n(V.n)
{
    // Исходный объект остается пустым, чтобы его деструктор не освободил чужую память
    V.p = nullptr;
    V.n = 0;
    cout << "  [Vector(Vector&&) - ПЕРЕМЕЩЕНИЕ n=" << n << "]" << endl;
}

// Деструктор: освобождает динамическую память
Vector::~Vector()
{
    cout << "  [~Vector() n=" << n << "]" << endl;
    delete[] p; // delete[] для nullptr безопасен
}

// Доступ к элементу по индексу (возвращаем ссылку, поэтому можно и читать, и записывать)
int &Vector::operator[](int index)
{
    return p[index];
}

const int &Vector::operator[](int index) const
{
    return p[index];
}

// Присваивание с копированием
Vector &Vector::operator=(const Vector &V)
{
    if (this != &V) // Защита от присваивания самому себе
    {
        delete[] p;           // Освобождаем память старого содержимого
        n = V.n;
        p = new int[n];       // Выделяем новую память
        for (int i = 0; i < n; i++)
            p[i] = V.p[i];    // Копируем элементы
    }
    cout << "  [operator=(const Vector&) - КОПИРОВАНИЕ]" << endl;
    return *this; // Возвращаем ссылку на текущий объект (допускает цепочку a = b = c)
}

// Присваивание с перемещением
Vector &Vector::operator=(Vector &&V) noexcept
{
    if (this != &V)
    {
        delete[] p;   // Освобождаем память старого содержимого
        p = V.p;      // Забираем указатель на чужой массив (без копирования элементов)
        n = V.n;
        V.p = nullptr; // Исходный вектор становится пустым
        V.n = 0;
    }
    cout << "  [operator=(Vector&&) - ПЕРЕМЕЩЕНИЕ]" << endl;
    return *this;
}

// Операция <: сравнение векторов последовательно по элементам
//  - элемент первого вектора меньше элемента второго  -> true
//  - элемент первого вектора больше элемента второго  -> false
//  - элементы равны                                    -> переход к следующей паре
//  - элементы в каком-либо векторе закончились         -> false
bool Vector::operator<(const Vector &V) const
{
    for (int i = 0; i < n && i < V.n; i++)
    {
        if (p[i] < V.p[i]) return true;
        if (p[i] > V.p[i]) return false;
    }
    return false;
}

// Вставка в поток: формат "n: (a0, a1, ...)"
ostream &operator<<(ostream &os, const Vector &V)
{
    os << V.n << ": (";
    for (int i = 0; i < V.n; i++)
    {
        if (i > 0) os << ", ";
        os << V.p[i];
    }
    os << ")";
    return os; // Возвращаем поток, чтобы можно было писать cout << a << b
}

// Извлечение из потока: сначала читается n, затем n целых чисел
istream &operator>>(istream &is, Vector &V)
{
    int k = 0;
    if (!(is >> k) || k < 0) return is; // Ошибка чтения - выходим (в потоке установлен флаг ошибки)
    int *tmp = new int[k];
    for (int i = 0; i < k; i++) is >> tmp[i];
    delete[] V.p;  // Освобождаем старое содержимое
    V.p = tmp;
    V.n = k;
    return is;
}

// part2.cpp - часть 2, вариант 22: наследование, статический и динамический полиморфизм
#include <iostream>
#include <vector>
using namespace std;

// Базовый класс - "вектор на плоскости"
class Vector2D
{
protected:            // Статус доступа protected: поля видны в производных классах
    double x, y;      // Координаты конца вектора (начало вектора в точке 0, 0)
public:
    Vector2D(double x, double y) : x(x), y(y) {} // Конструктор для инициализации полей
    virtual ~Vector2D() {}                       // Виртуальный деструктор (удаление через указатель на базовый класс)

    // Обычная (не виртуальная) функция печати: СТАТИЧЕСКОЕ связывание
    void printStatic() const
    {
        cout << "Vector2D: x=" << x << " y=" << y << endl;
    }

    // Виртуальная функция печати: ДИНАМИЧЕСКОЕ связывание
    virtual void print() const
    {
        cout << "Vector2D: x=" << x << " y=" << y << endl;
    }
};

// Производный класс - "вектор в трехмерном пространстве"
class Vector3D : public Vector2D
{
    double z; // Дополнительная координата
public:
    Vector3D(double x, double y, double z) : Vector2D(x, y), z(z) {} // Явный вызов конструктора базового класса

    // Переопределенная обычная функция (статическое связывание)
    void printStatic() const
    {
        cout << "Vector3D: ";
        Vector2D::printStatic();   // Сначала вызывается функция базового класса
        cout << "          z=" << z << endl;
    }

    // Переопределенная виртуальная функция (динамическое связывание)
    void print() const override
    {
        cout << "Vector3D: ";
        Vector2D::print();         // Сначала вызывается функция базового класса
        cout << "          z=" << z << endl;
    }
};

int main()
{
    Vector2D v2(1, 2);     // По одному объекту каждого класса
    Vector3D v3(3, 4, 5);

    cout << "=== 1. Вызовы через сами объекты ===" << endl;
    v2.print();
    v3.print();

    cout << "\n=== 2. Статический полиморфизм (функция НЕ virtual) ===" << endl;
    Vector2D *p = &v2;
    cout << "p -> объект Vector2D:" << endl;
    p->printStatic();
    p = &v3; // Указатель на базовый класс ссылается на объект производного класса
    cout << "p -> объект Vector3D (вызов определяется ТИПОМ УКАЗАТЕЛЯ):" << endl;
    p->printStatic(); // Вызовется функция Vector2D, координата z не печатается

    cout << "\n=== 3. Динамический полиморфизм (функция virtual) ===" << endl;
    p = &v2;
    cout << "p -> объект Vector2D:" << endl;
    p->print();
    p = &v3;
    cout << "p -> объект Vector3D (вызов определяется ТИПОМ ОБЪЕКТА):" << endl;
    p->print(); // Вызовется функция Vector3D, печатается и z

    cout << "\n=== 4. Вектор указателей на объекты разных классов ===" << endl;
    vector<Vector2D *> arr;               // Один контейнер - указатели на базовый класс
    arr.push_back(new Vector2D(1, 2));
    arr.push_back(new Vector3D(3, 4, 5));
    arr.push_back(new Vector2D(-1.5, 0));
    arr.push_back(new Vector3D(0, -2, 7.5));

    for (size_t i = 0; i < arr.size(); i++)
    {
        cout << "Элемент " << i << ":" << endl;
        arr[i]->print(); // Для каждого объекта вызывается "свой" вариант print
    }

    for (Vector2D *q : arr) delete q; // Освобождаем память (работает благодаря virtual ~Vector2D)
    return 0;
}
